option(VOD_WITH_ETCD "Build etcd service discovery" ON)
if(VOD_WITH_ETCD)
    find_package(OpenSSL REQUIRED)
    find_path(ETCD_INCLUDE_DIR etcd/Client.hpp REQUIRED)
    find_library(ETCD_LIBRARY etcd-cpp-api REQUIRED)
    find_library(CPPREST_LIBRARY cpprest REQUIRED)
    add_library(vod_etcd src/rpc/etcd_discovery.cpp src/rpc/etcd_registration.cpp)
    target_include_directories(vod_etcd PUBLIC include PRIVATE ${ETCD_INCLUDE_DIR})
    target_link_libraries(vod_etcd PUBLIC vod_rpc PRIVATE ${ETCD_LIBRARY} ${CPPREST_LIBRARY} OpenSSL::SSL OpenSSL::Crypto)
    find_package(Protobuf REQUIRED)
    set(GFLAGS_USE_TARGET_NAMESPACE TRUE)
    find_package(gflags REQUIRED)
    add_executable(vod_transcode_worker src/transcode/worker.cpp)
    target_link_libraries(vod_transcode_worker PRIVATE vod_job_store vod_hls_lib vod_file_store vod_mq gflags::gflags)
    protobuf_generate_cpp(HEALTH_PROTO_SOURCES HEALTH_PROTO_HEADERS proto/health.proto)
    add_library(vod_health_protocol ${HEALTH_PROTO_SOURCES} ${HEALTH_PROTO_HEADERS})
    target_include_directories(vod_health_protocol PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(vod_health_protocol PUBLIC protobuf::libprotobuf)
    foreach(role IN ITEMS server client)
        add_executable(vod_health_${role} src/rpc/health_${role}.cpp)
        target_link_libraries(vod_health_${role} PRIVATE vod_etcd vod_health_protocol
            gflags::gflags brpc OpenSSL::SSL OpenSSL::Crypto leveldb z dl)
    endforeach()
    find_package(jsoncpp CONFIG REQUIRED)
    protobuf_generate_cpp(USER_PROTO_SOURCES USER_PROTO_HEADERS proto/user.proto)
    add_library(vod_user_protocol ${USER_PROTO_SOURCES} ${USER_PROTO_HEADERS})
    target_include_directories(vod_user_protocol PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(vod_user_protocol PUBLIC protobuf::libprotobuf)
    add_executable(vod_user_server src/user/user_server.cpp)
    target_link_libraries(vod_user_server PRIVATE vod_etcd vod_user_protocol vod_session_store
        gflags::gflags brpc OpenSSL::SSL OpenSSL::Crypto leveldb z dl)
    add_executable(vod_gateway src/gateway/main.cpp)
    target_link_libraries(vod_gateway PRIVATE vod_etcd vod_user_protocol JsonCpp::JsonCpp
        gflags::gflags brpc OpenSSL::SSL OpenSSL::Crypto leveldb z dl)
    protobuf_generate_cpp(VIDEO_PROTO_SOURCES VIDEO_PROTO_HEADERS proto/video.proto)
    add_library(vod_video_protocol ${VIDEO_PROTO_SOURCES} ${VIDEO_PROTO_HEADERS})
    target_include_directories(vod_video_protocol PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(vod_video_protocol PUBLIC protobuf::libprotobuf)
    add_executable(vod_video_server src/video/video_server.cpp)
    target_link_libraries(vod_video_server PRIVATE vod_etcd vod_video_protocol vod_video_store vod_mq
        gflags::gflags brpc OpenSSL::SSL OpenSSL::Crypto leveldb z dl)
    target_link_libraries(vod_gateway PRIVATE vod_video_protocol)
    add_executable(vod_file_server src/file/file_server.cpp)
    target_link_libraries(vod_file_server PRIVATE vod_file_store vod_etcd JsonCpp::JsonCpp gflags::gflags)
endif()
