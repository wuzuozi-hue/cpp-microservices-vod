find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_path(MPV_INCLUDE_DIR mpv/client.h REQUIRED)
find_library(MPV_LIBRARY NAMES mpv libmpv REQUIRED)
set(CMAKE_AUTOMOC ON)
add_executable(vod_player main.cpp)
target_include_directories(vod_player PRIVATE ${MPV_INCLUDE_DIR})
target_link_libraries(vod_player PRIVATE Qt6::Widgets Qt6::Network ${MPV_LIBRARY})
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_test(NAME vod_player_smoke COMMAND vod_player --self-test)
    set_tests_properties(vod_player_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME vod_player_playback COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/run_qt_player_test.py $<TARGET_FILE:vod_player>)
    add_executable(vod_client_ui_test ${CMAKE_SOURCE_DIR}/tests/client_ui_test.cpp)
    target_include_directories(vod_client_ui_test PRIVATE ${MPV_INCLUDE_DIR})
    target_link_libraries(vod_client_ui_test PRIVATE Qt6::Widgets Qt6::Network Qt6::Test ${MPV_LIBRARY})
    add_test(NAME vod_client_ui_workflow COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/run_client_ui_test.py $<TARGET_FILE:vod_client_ui_test>)
endif()
